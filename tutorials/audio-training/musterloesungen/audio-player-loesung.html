Vollständige Implementierung mit Web Audio API

Audio-Player Implementierung

Dies ist die vollständige Musterlösung für den Audio-Player mit Web Audio API. Der Player bietet alle grundlegenden Funktionen sowie einige erweiterte Effekte.

  <script>
    class AudioPlayer {
      constructor() {
        // Audio-Kontext erstellen
        this.audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        
        // Audio-Elemente
        this.audioFileInput = document.getElementById('audioFile');
        this.playBtn = document.getElementById('playBtn');
        this.pauseBtn = document.getElementById('pauseBtn');
        this.stopBtn = document.getElementById('stopBtn');
        this.loadDemoBtn = document.getElementById('loadDemoBtn');
        this.currentTimeSpan = document.getElementById('currentTime');
        this.totalTimeSpan = document.getElementById('totalTime');
        this.progressBar = document.getElementById('progressBar');
        
        // Effekt-Elemente
        this.volumeSlider = document.getElementById('volumeSlider');
        this.volumeValue = document.getElementById('volumeValue');
        this.filterType = document.getElementById('filterType');
        this.filterFreq = document.getElementById('filterFreq');
        this.filterFreqValue = document.getElementById('filterFreqValue');
        this.delayTime = document.getElementById('delayTime');
        this.delayValue = document.getElementById('delayValue');
        
        // Audio-Buffer und -Quelle
        this.audioBuffer = null;
        this.source = null;
        this.isPlaying = false;
        this.startTime = 0;
        this.pauseTime = 0;
        
        // Effekt-Nodes
        this.gainNode = null;
        this.filterNode = null;
        this.delayNode = null;
        this.feedbackNode = null;
        
        // Event-Listener
        this.setupEventListeners();
        
        // Effekte initialisieren
        this.initEffects();
      }
      
      setupEventListeners() {
        // Wiedergabesteuerung
        this.playBtn.addEventListener('click', () => this.play());
        this.pauseBtn.addEventListener('click', () => this.pause());
        this.stopBtn.addEventListener('click', () => this.stop());
        
        // Dateiauswahl
        this.audioFileInput.addEventListener('change', (e) => this.handleFileSelect(e));
        
        // Demo-Button
        this.loadDemoBtn.addEventListener('click', () => this.loadDemoAudio());
        
        // Effekt-Steuerung
        this.volumeSlider.addEventListener('input', (e) => this.setVolume(e.target.value));
        this.filterType.addEventListener('change', () => this.setFilterType());
        this.filterFreq.addEventListener('input', (e) => this.setFilterFreq(e.target.value));
        this.delayTime.addEventListener('input', (e) => this.setDelayTime(e.target.value));
        
        // Fortschrittsaktualisierung
        setInterval(() => this.updateProgress(), 100);
      }
      
      initEffects() {
        // Gain-Node für Lautstärke
        this.gainNode = this.audioCtx.createGain();
        this.gainNode.gain.value = this.volumeSlider.value;
        
        // Filter-Node
        this.filterNode = this.audioCtx.createBiquadFilter();
        this.filterNode.type = this.filterType.value;
        this.filterNode.frequency.value = this.filterFreq.value;
        
        // Delay-Node
        this.delayNode = this.audioCtx.createDelay(1.0);
        this.feedbackNode = this.audioCtx.createGain();
        this.feedbackNode.gain.value = 0.3;
        
        // Verbindungen: Quelle -> Gain -> Filter -> Delay -> Feedback -> Destination
        this.gainNode.connect(this.filterNode);
        this.filterNode.connect(this.delayNode);
        this.delayNode.connect(this.feedbackNode);
        this.feedbackNode.connect(this.delayNode); // Feedback-Schleife
        this.delayNode.connect(this.audioCtx.destination);
      }
      
      async handleFileSelect(e) {
        const file = e.target.files[0];
        if (!file) return;
        
        const arrayBuffer = await file.arrayBuffer();
        this.audioBuffer = await this.audioCtx.decodeAudioData(arrayBuffer);
        this.updateTimeDisplay();
      }
      
      async loadDemoAudio() {
        try {
          // In einer echten Implementierung würde hier eine Demo-Audiodatei geladen
          // Für dieses Beispiel erzeugen wir einen Oszillator
          this.createDemoTone();
        } catch (error) {
          console.error('Fehler beim Laden der Demo-Audio:', error);
        }
      }
      
      createDemoTone() {
        // Erstelle einen Demo-Ton (Sinuswelle)
        this.demoSource = this.audioCtx.createOscillator();
        this.demoSource.type = 'sine';
        this.demoSource.frequency.value = 440; // A4 Note
        
        // Verbinde mit Effekt-Chain
        this.demoSource.connect(this.gainNode);
        
        // Starte den Oszillator
        this.demoSource.start();
        
        // Setze Flag für Demo-Modus
        this.isDemoMode = true;
      }
      
      play() {
        if (this.isDemoMode) {
          // In Demo-Modus starte den Oszillator
          if (this.audioCtx.state === 'suspended') {
            this.audioCtx.resume();
          }
          this.isPlaying = true;
          this.startTime = this.audioCtx.currentTime;
          return;
        }
        
        if (!this.audioBuffer) {
          alert('Bitte wähle zuerst eine Audiodatei aus oder lade die Demo!');
          return;
        }
        
        if (this.audioCtx.state === 'suspended') {
          this.audioCtx.resume();
        }
        
        // Stoppe vorherige Quelle falls vorhanden
        if (this.source) {
          this.source.stop();
        }
        
        // Erstelle neue Quelle
        this.source = this.audioCtx.createBufferSource();
        this.source.buffer = this.audioBuffer;
        this.source.connect(this.gainNode);
        
        // Starte Wiedergabe
        this.source.start(0, this.pauseTime);
        this.isPlaying = true;
        this.startTime = this.audioCtx.currentTime - this.pauseTime;
      }
      
      pause() {
        if (!this.isPlaying) return;
        
        if (this.isDemoMode) {
          // Für Demo-Modus pausieren wir den Oszillator
          this.demoSource.stop();
          this.isPlaying = false;
          this.createDemoTone(); // Erstelle neuen Oszillator für späteres Fortsetzen
          return;
        }
        
        // Speichere aktuelle Position
        this.pauseTime = this.audioCtx.currentTime - this.startTime;
        
        // Stoppe aktuelle Quelle
        if (this.source) {
          this.source.stop();
          this.source = null;
        }
        
        this.isPlaying = false;
      }
      
      stop() {
        if (this.source) {
          this.source.stop();
          this.source = null;
        }
        
        if (this.demoSource) {
          this.demoSource.stop();
          this.demoSource = null;
        }
        
        this.isPlaying = false;
        this.pauseTime = 0;
        this.startTime = 0;
        this.progressBar.style.width = '0%';
        this.currentTimeSpan.textContent = '0:00';
      }
      
      setVolume(value) {
        if (this.gainNode) {
          this.gainNode.gain.value = value;
        }
        this.volumeValue.textContent = Math.round(value * 100) + '%';
      }
      
      setFilterType() {
        if (this.filterNode) {
          this.filterNode.type = this.filterType.value;
        }
      }
      
      setFilterFreq(value) {
        if (this.filterNode) {
          this.filterNode.frequency.value = value;
        }
        this.filterFreqValue.textContent = value + ' Hz';
      }
      
      setDelayTime(value) {
        if (this.delayNode) {
          this.delayNode.delayTime.value = value;
        }
        this.delayValue.textContent = value + ' s';
      }
      
      updateProgress() {
        if (!this.isPlaying) return;
        
        let currentTime = 0;
        if (this.isDemoMode) {
          // In Demo-Modus zeigen wir eine fiktive Zeit an
          currentTime = (this.audioCtx.currentTime - this.startTime) % 10; // Zyklus von 10 Sekunden
        } else {
          currentTime = this.audioCtx.currentTime - this.startTime;
        }
        
        // Aktualisiere Fortschrittsbalken
        if (this.audioBuffer && !this.isDemoMode) {
          const progress = (currentTime / this.audioBuffer.duration) * 100;
          this.progressBar.style.width = Math.min(progress, 100) + '%';
        } else {
          // In Demo-Modus füllen wir den Balken zyklisch
          this.progressBar.style.width = ((currentTime % 10) / 10) * 100 + '%';
        }
        
        // Aktualisiere Zeit-Anzeige
        this.currentTimeSpan.textContent = this.formatTime(currentTime);
      }
      
      updateTimeDisplay() {
        if (this.audioBuffer) {
          this.totalTimeSpan.textContent = this.formatTime(this.audioBuffer.duration);
        }
      }
      
      formatTime(seconds) {
        const mins = Math.floor(seconds / 60);
        const secs = Math.floor(seconds % 60);
        return mins + ':' + (secs < 10 ? '0' : '') + secs;
      }
    }
    
    // Initialisiere den Player wenn die Seite geladen ist
    document.addEventListener('DOMContentLoaded', () => {
      new AudioPlayer();
    });
  </script>
</body>
</html>

Erklärung der Implementierung

1. AudioContext-Initialisierung

Der AudioContext ist das Herzstück der Web Audio API. Er wird einmalig erstellt und für alle Audio-Operationen verwendet: 

this.audioCtx = new (window.AudioContext || window.webkitAudioContext)();

2. Audio-Graph-Struktur

Der Audio-Graph verbindet verschiedene Nodes in folgender Reihenfolge: 

Quelle → Gain → Filter → Delay → Feedback → Destination

3. Effekt-Implementierung

  • GainNode: Für Lautstärkeregelung
  • BiquadFilterNode: Für Frequenzfilterung (Lowpass, Highpass, Bandpass)
  • DelayNode: Für Echo-Effekte
  • Feedback: Um Verzögerung zu wiederholen

Wichtige Funktionen

Wiedergabesteuerung

Die play(), pause() und stop() Funktionen steuern die Audio-Wiedergabe: 

play() {
  if (this.audioCtx.state === 'suspended') {
    this.audioCtx.resume();  // Benötigt Benutzerinteraktion
  }
  
  // Erstelle neue Quelle und starte Wiedergabe
  this.source = this.audioCtx.createBufferSource();
  this.source.buffer = this.audioBuffer;
  this.source.connect(this.gainNode);
  this.source.start(0, this.pauseTime);
}

Effekt-Steuerung

Die Effekte werden in Echtzeit gesteuert: 

setFilterFreq(value) {
  if (this.filterNode) {
    this.filterNode.frequency.value = value;  // Frequenz in Hz
  }
}

Erweiterungsmöglichkeiten

1. Visualisierung hinzufügen

Mit der AnalyserNode können Frequenz- und Zeitdaten extrahiert werden: 

const analyser = this.audioCtx.createAnalyser();
analyser.fftSize = 2048;
this.gainNode.connect(analyser);
analyser.connect(this.audioCtx.destination);

// Abfrage der Frequenzdaten
const bufferLength = analyser.frequencyBinCount;
const dataArray = new Uint8Array(bufferLength);
analyser.getByteFrequencyData(dataArray);

2. Equalizer implementieren

Mehrere BiquadFilterNodes können kombiniert werden, um einen Equalizer zu erstellen: 

const bands = [80, 200, 500, 1000, 2000, 5000, 10000];
const filters = [];

for (let i = 0; i < bands.length; i++) {
  const filter = this.audioCtx.createBiquadFilter();
  filter.type = 'peaking';
  filter.frequency.value = bands[i];
  filter.Q.value = 1.0;
  filter.gain.value = 0;  // 0dB = keine Verstärkung
  
  filters.push(filter);
}

3. Playlist-Funktionalität

Implementierung einer Wiedergabeliste mit mehreren Tracks: 

class Playlist {
  constructor(player) {
    this.player = player;
    this.tracks = [];
    this.currentTrackIndex = 0;
  }
  
  addTrack(track) {
    this.tracks.push(track);
  }
  
  nextTrack() {
    if (this.currentTrackIndex < this.tracks.length - 1) {
      this.currentTrackIndex++;
      this.playCurrentTrack();
    }
  }
  
  previousTrack() {
    if (this.currentTrackIndex > 0) {
      this.currentTrackIndex--;
      this.playCurrentTrack();
    }
  }
}

Best Practices

Performance

  • AudioContext erst nach Benutzerinteraktion starten
  • Effekte nur aktivieren, wenn benötigt
  • Keine komplexen Operationen im Hauptthread
  • Effizientes Handling von Audio-Buffer-Objekten

Sicherheit

  • Keine direkte Verarbeitung von Benutzerdateien ohne Validierung
  • HTTPS für Web Audio API Funktionen verwenden
  • Respektiere Urheberrechte bei Audio-Dateien

Barrierefreiheit

  • Sinnvolle ARIA-Labels für Steuerelemente
  • Tastaturnavigation unterstützen
  • Kontraststarke Farben verwenden
  • Alternativen für visuelle Effekte bieten

Debugging-Tipps

Browser-Entwicklungswerkzeuge

  • Verwende die Audio-Tab in Chrome DevTools
  • Überprüfe den Audio-Graph in der Console
  • Prüfe auf Fehlermeldungen in der Console
  • Überprüfe die Sample-Rate und Buffer-Größe
